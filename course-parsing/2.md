## 💡 Как сайт устроен

Страницы выглядят так:

```
http://quotes.toscrape.com/page/1
http://quotes.toscrape.com/page/2
http://quotes.toscrape.com/page/3
...
```

Внизу каждой страницы есть кнопка "Next →", если есть следующая страница.

---

## 🧠 Что научимся:

* Искать ссылку "Следующая страница"
* Подставлять ссылку в цикл
* Автоматически прыгать по страницам, пока не кончатся

---

## 🚀 Полный код второго урока:

```python
import requests  # Библиотека для HTTP-запросов
from bs4 import BeautifulSoup  # Для разбора HTML-страниц

base_url = "http://quotes.toscrape.com"  # Основной адрес сайта
page_url = "/page/1"  # Первая страница сайта с цитатами
all_quotes = []  # Здесь будем хранить все собранные цитаты

# Пока есть ссылка на следующую страницу — продолжаем парсинг
while page_url:
    response = requests.get(base_url + page_url)  # Делаем запрос к текущей странице
    soup = BeautifulSoup(response.text, "lxml")  # Преобразуем HTML в объект BeautifulSoup

    quotes = soup.find_all("div", class_="quote")  # Находим все блоки с цитатами на странице

    for quote in quotes:
        text = quote.find("span", class_="text").text  # Извлекаем саму цитату
        author = quote.find("small", class_="author").text  # Извлекаем имя автора
        tags = [tag.text for tag in quote.find_all("a", class_="tag")]  # Собираем все теги цитаты

        all_quotes.append({  # Сохраняем всё в виде словаря
            "text": text,
            "author": author,
            "tags": tags
        })

    next_btn = soup.find("li", class_="next")  # Ищем кнопку "Next" (если есть)

    if next_btn:  # Если она есть — переходим по ссылке
        page_url = next_btn.find("a")["href"]  # Вытаскиваем относительный путь, например: "/page/2"
    else:
        page_url = None  # Если кнопки нет — мы на последней странице

# Выводим все собранные цитаты
for q in all_quotes:
    print("📜", q["text"])      # Цитата
    print("✍️", q["author"])    # Автор
    print("🏷️", q["tags"])      # Список тегов
    print("-" * 50)             # Разделитель
```

## Код с ограничителем по страницам

import requests  # Библиотека для отправки HTTP-запросов
from bs4 import BeautifulSoup  # HTML-парсер для удобной навигации по тегам

base_url = "http://quotes.toscrape.com"  # Базовый адрес сайта
page_url = "/page/1"  # Относительный путь к первой странице
all_quotes = []  # Сюда будем сохранять все цитаты

max_pages = 10  # Максимальное количество страниц, которое разрешено обойти
page_count = 0  # Счётчик текущей страницы

# Основной цикл: пока есть следующая страница и не превышен лимит

```python
while page_url and page_count < max_pages:
    response = requests.get(base_url + page_url)  # Отправляем запрос к текущей странице
    soup = BeautifulSoup(response.text, "lxml")  # Превращаем HTML в объект BeautifulSoup

    quotes = soup.find_all("div", class_="quote")  # Ищем все блоки цитат на странице

    for quote in quotes:
        text = quote.find("span", class_="text").text  # Извлекаем текст цитаты
        author = quote.find("small", class_="author").text  # Имя автора
        tags = [tag.text for tag in quote.find_all("a", class_="tag")]  # Все теги как список

        all_quotes.append({  # Сохраняем данные одной цитаты в список
            "text": text,
            "author": author,
            "tags": tags
        })

    next_btn = soup.find("li", class_="next")  # Ищем кнопку перехода на следующую страницу

    if next_btn:
        page_url = next_btn.find("a")["href"]  # Получаем путь к следующей странице (например, "/page/2")
    else:
        page_url = None  # Кнопки "Next" нет — это последняя страница

    page_count += 1  # Увеличиваем счётчик страниц

# Выводим все собранные цитаты
for q in all_quotes:
    print("📜", q["text"])       # Текст цитаты
    print("✍️", q["author"])     # Имя автора
    print("🏷️", q["tags"])       # Список тегов
    print("-" * 50)             # Разделитель
```
